WITH STATS AS ( SELECT MIN(MILLISECONDS) AS MIN_MS, AVG(MILLISECONDS) AS AVG_MS, MAX(MILLISECONDS) AS MAX_MS FROM TRACK ), BOUNDS AS ( SELECT (MIN_MS + AVG_MS) / 2.0 AS MID_MIN_AVG, (AVG_MS + MAX_MS) / 2.0 AS MID_AVG_MAX FROM STATS ), TRACK_CAT AS ( SELECT T.TRACKID, T.MILLISECONDS AS MS, CASE WHEN T.MILLISECONDS < B.MID_MIN_AVG THEN 'SHORT' WHEN T.MILLISECONDS < B.MID_AVG_MAX THEN 'MEDIUM' ELSE 'LONG' END AS CATEGORY FROM TRACK T CROSS JOIN BOUNDS B ), REVENUE_BY_TRACK AS ( SELECT IL.TRACKID, SUM(IL.UNITPRICE * IL.QUANTITY) AS REVENUE FROM INVOICELINE IL GROUP BY IL.TRACKID ) SELECT TC.CATEGORY, MIN(TC.MS) / 60000.0 AS MIN_MINUTES, MAX(TC.MS) / 60000.0 AS MAX_MINUTES, SUM(COALESCE(RBT.REVENUE, 0)) AS TOTAL_REVENUE FROM TRACK_CAT TC LEFT JOIN REVENUE_BY_TRACK RBT ON RBT.TRACKID = TC.TRACKID GROUP BY TC.CATEGORY ORDER BY CASE TC.CATEGORY WHEN 'SHORT' THEN 1 WHEN 'MEDIUM' THEN 2 ELSE 3 END;
