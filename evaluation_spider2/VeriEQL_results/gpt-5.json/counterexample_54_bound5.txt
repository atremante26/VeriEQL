WITH DELIVERED_ORDERS AS ( SELECT ORDER_ID, STRFTIME('%Y', ORDER_DELIVERED_CUSTOMER_DATE) AS YEAR, STRFTIME('%M', ORDER_DELIVERED_CUSTOMER_DATE) AS MONTH FROM OLIST_ORDERS WHERE ORDER_STATUS = 'DELIVERED' AND ORDER_DELIVERED_CUSTOMER_DATE IS NOT NULL AND STRFTIME('%Y', ORDER_DELIVERED_CUSTOMER_DATE) IN ('2016', '2017', '2018') ), ANNUAL_COUNTS AS ( SELECT YEAR, COUNT(*) AS ANNUAL_DELIVERED_ORDERS FROM DELIVERED_ORDERS GROUP BY YEAR ), LOWEST_YEAR AS ( SELECT YEAR FROM ANNUAL_COUNTS ORDER BY ANNUAL_DELIVERED_ORDERS ASC LIMIT 1 ), MONTHLY_COUNTS_LOWEST_YEAR AS ( SELECT MONTH, COUNT(*) AS MONTHLY_DELIVERED_ORDERS FROM DELIVERED_ORDERS WHERE YEAR = (SELECT YEAR FROM LOWEST_YEAR) GROUP BY MONTH ) SELECT MAX(MONTHLY_DELIVERED_ORDERS) AS HIGHEST_MONTHLY_DELIVERED_ORDERS FROM MONTHLY_COUNTS_LOWEST_YEAR;

-- 2
WITH DELIVERED AS ( SELECT ORDER_ID, STRFTIME('%Y', ORDER_DELIVERED_CUSTOMER_DATE) AS Y, STRFTIME('%M', ORDER_DELIVERED_CUSTOMER_DATE) AS M FROM OLIST_ORDERS WHERE ORDER_STATUS = 'DELIVERED' AND ORDER_DELIVERED_CUSTOMER_DATE IS NOT NULL AND STRFTIME('%Y', ORDER_DELIVERED_CUSTOMER_DATE) IN ('2016','2017','2018') ), ANNUAL AS ( SELECT Y, COUNT(*) AS ANNUAL_COUNT FROM DELIVERED GROUP BY Y ), MIN_YEAR AS ( SELECT Y FROM ANNUAL WHERE ANNUAL_COUNT = (SELECT MIN(ANNUAL_COUNT) FROM ANNUAL) ), MONTHLY AS ( SELECT Y, M, COUNT(*) AS MONTHLY_COUNT FROM DELIVERED WHERE Y IN (SELECT Y FROM MIN_YEAR) GROUP BY Y, M ) SELECT MAX(MONTHLY_COUNT) AS HIGHEST_MONTHLY_DELIVERED_ORDERS FROM MONTHLY;
