SELECT SCHOOLS.SCHOOL, (FRPM.FRPM_COUNT_AGES_5_17 / FRPM.ENROLLMENT_AGES_5_17) * 100 AS FRPM_PERCENT FROM SCHOOLS INNER JOIN FRPM ON SCHOOLS.CDSCODE = FRPM.CDSCODE WHERE SCHOOLS.COUNTY = 'LOS ANGELES' AND SCHOOLS.GSOFFERED = 'K-9';
