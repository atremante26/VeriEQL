-- ----------sql1------------
SELECT (CAST(STRFTIME('%Y', E.EXAMINATION_DATE) AS INTEGER) - CAST(STRFTIME('%Y', P.BIRTHDAY) AS INTEGER)) AS AGE, P.DIAGNOSIS FROM LABORATORY L INNER JOIN PATIENT P ON L.ID = P.ID INNER JOIN EXAMINATION E ON L.ID = E.ID WHERE L.HGB = (SELECT MAX(HGB) FROM LABORATORY);
