SELECT (SUM(CASE WHEN POSITION = 'MEMBER' AND MAJOR_NAME = 'BUSINESS' THEN 1 ELSE 0 END) * 100.0 / COUNT(MEMBER_ID)) AS BUSINESS_MAJOR_PERCENTAGE FROM MEMBER JOIN MAJOR ON MEMBER.LINK_TO_MAJOR = MAJOR.MAJOR_ID;
-- ----------sql2------------
-- NULL
SELECT CAST(SUM(CASE WHEN T2.MAJOR_NAME = 'BUSINESS' THEN 1 ELSE 0 END) AS REAL) * 100 / COUNT(T1.MEMBER_ID) FROM MEMBER AS T1 INNER JOIN MAJOR AS T2 ON T2.MAJOR_ID = T1.LINK_TO_MAJOR WHERE T1.POSITION = 'MEMBER';
