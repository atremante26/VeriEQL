-- NOAH SNYDER
SELECT DISPLAYNAME FROM ( SELECT U.DISPLAYNAME, SUM(P.VIEWCOUNT) AS TOTALVIEWCOUNT FROM USERS U JOIN POSTS P ON U.ID = P.OWNERUSERID WHERE U.DISPLAYNAME IN ('HARVEY MOTULSKY', 'NOAH SNYDER') GROUP BY U.DISPLAYNAME ORDER BY TOTALVIEWCOUNT DESC LIMIT 1 ) AS HIGHERPOPULARITYUSER;

-- HARVEY MOTULSKY
SELECT T1.DISPLAYNAME FROM USERS AS T1 INNER JOIN POSTHISTORY AS T2 ON T1.ID = T2.USERID INNER JOIN POSTS AS T3 ON T2.POSTID = T3.ID WHERE T1.DISPLAYNAME = 'HARVEY MOTULSKY' OR T1.DISPLAYNAME = 'NOAH SNYDER' GROUP BY T1.DISPLAYNAME ORDER BY SUM(T3.VIEWCOUNT) DESC LIMIT 1;
