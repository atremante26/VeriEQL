SELECT Z.CITY, Z.COUNTY FROM INCOME INNER JOIN EVENT ON INCOME.LINK_TO_MEMBER = EVENT.EVENT_ID INNER JOIN ZIP_CODE Z ON EVENT.LOCATION = Z.ZIP_CODE GROUP BY Z.CITY, Z.COUNTY HAVING COUNT(INCOME.AMOUNT) > 40;
