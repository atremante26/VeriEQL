SELECT COUNT(DISTINCT SCHOOLS.CDSCODE) FROM SCHOOLS INNER JOIN SATSCORES ON SCHOOLS.CDSCODE = SATSCORES.CDS INNER JOIN FRPM ON SCHOOLS.CDSCODE = FRPM.CDSCODE WHERE SATSCORES.AVGSCRMATH > 560 AND FRPM.CHARTER_FUNDING_TYPE = 'DIRECTLY FUNDED';
