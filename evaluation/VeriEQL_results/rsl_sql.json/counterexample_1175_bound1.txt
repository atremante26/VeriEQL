	ID INTEGER,

SELECT STRFTIME('%Y', EXAMINATION.EXAMINATION_DATE) - STRFTIME('%Y', PATIENT.BIRTHDAY) AS AGE, EXAMINATION.DIAGNOSIS FROM PATIENT INNER JOIN LABORATORY ON PATIENT.ID = LABORATORY.ID INNER JOIN EXAMINATION ON PATIENT.ID = EXAMINATION.ID WHERE LABORATORY.HGB = (SELECT MAX(HGB) FROM LABORATORY WHERE LABORATORY.ID = PATIENT.ID) LIMIT 1;
