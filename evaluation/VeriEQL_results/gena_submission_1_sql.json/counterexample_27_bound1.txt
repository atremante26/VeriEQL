CREATE TABLE SCHOOLS (
	CDSCODE VARCHAR(20),
	SCHOOL VARCHAR(20),
	PHONE VARCHAR(20),
	OPENDATE DATE,
	CLOSEDDATE DATE
);
INSERT INTO SCHOOLS VALUES ('', '', NULL, '4000-03-01', '4200-04-01');
-- ----------sql1------------
-- A, 0, NULL
SELECT T1.SNAME,T1.AVGSCRWRITE,T2.PHONE FROM SATSCORES AS T1 INNER JOIN SCHOOLS AS T2 ON T1.CDS = T2.CDSCODE WHERE STRFTIME('%Y', T2.OPENDATE) > '1991' OR STRFTIME('%Y', T2.CLOSEDDATE) < '2000';
-- ----------sql2------------
-- , 0, NULL
SELECT T2.SCHOOL, T1.AVGSCRWRITE, T2.PHONE FROM SCHOOLS AS T2 LEFT JOIN SATSCORES AS T1 ON T2.CDSCODE = T1.CDS WHERE STRFTIME('%Y', T2.OPENDATE) > '1991' OR STRFTIME('%Y', T2.CLOSEDDATE) < '2000';
