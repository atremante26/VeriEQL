SELECT T1.SCHOOL, T1.FUNDINGTYPE FROM SCHOOLS AS T1 INNER JOIN ( SELECT CDS, AVG(AVGSCRMATH) AS AVG_MATH FROM SATSCORES GROUP BY CDS HAVING AVG(AVGSCRMATH) > 400 ) AS T2 ON T1.CDSCODE = T2.CDS WHERE T1.CITY = 'RIVERSIDE';
