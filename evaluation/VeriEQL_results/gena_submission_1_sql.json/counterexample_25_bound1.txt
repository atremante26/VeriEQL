CREATE TABLE FRPM (
	CDSCODE VARCHAR(20),
	DISTRICT_NAME VARCHAR(20),
	CHARTER_FUNDING_TYPE VARCHAR(20)
);
INSERT INTO FRPM VALUES ('', 'RIVERSIDE', '');
CREATE TABLE SATSCORES (
	CDS VARCHAR(20),
	SNAME VARCHAR(20),
	AVGSCRMATH INTEGER
);
INSERT INTO SATSCORES VALUES ('', NULL, 401);
CREATE TABLE SCHOOLS (
	CDSCODE VARCHAR(20),
	DISTRICT VARCHAR(20),
	SCHOOL VARCHAR(20),
	CITY VARCHAR(20),
	CHARTER INTEGER,
	FUNDINGTYPE VARCHAR(20)
);
INSERT INTO SCHOOLS VALUES ('', '', '', 'A', -2147483648, '');
-- ----------sql1------------
SELECT T1.SCHOOL, T1.FUNDINGTYPE FROM SCHOOLS AS T1 INNER JOIN ( SELECT CDS, AVG(AVGSCRMATH) AS AVG_MATH FROM SATSCORES GROUP BY CDS HAVING AVG(AVGSCRMATH) > 400 ) AS T2 ON T1.CDSCODE = T2.CDS WHERE T1.CITY = 'RIVERSIDE';

-- NULL, 
SELECT T1.SNAME, T2.CHARTER_FUNDING_TYPE FROM SATSCORES AS T1 INNER JOIN FRPM AS T2 ON T1.CDS = T2.CDSCODE WHERE T2.DISTRICT_NAME LIKE 'RIVERSIDE%' GROUP BY T1.SNAME, T2.CHARTER_FUNDING_TYPE HAVING CAST(SUM(T1.AVGSCRMATH) AS REAL) / COUNT(T1.CDS) > 400;
