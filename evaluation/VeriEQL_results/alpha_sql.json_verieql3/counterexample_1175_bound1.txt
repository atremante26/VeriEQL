CREATE TABLE EXAMINATION (
	ID INTEGER,
	EXAMINATION_DATE DATE,
	DIAGNOSIS VARCHAR(20)
);
INSERT INTO EXAMINATION VALUES (0, '2000-07-01', 'A');
CREATE TABLE PATIENT (
	ID INTEGER,
	BIRTHDAY DATE,
	DIAGNOSIS VARCHAR(20)
);
INSERT INTO PATIENT VALUES (0, '4000-06-01', '');
CREATE TABLE LABORATORY (
	ID INTEGER,
	DATE DATE,
	HGB INTEGER,
	RA VARCHAR(20),
	RF VARCHAR(20)
);
INSERT INTO LABORATORY VALUES (0, '4000-01-01', 34, '', '');
-- ----------sql1------------
-- -2000, A
SELECT STRFTIME('%Y', EXAMINATION.EXAMINATION_DATE) - STRFTIME('%Y', PATIENT.BIRTHDAY) AS AGE, EXAMINATION.DIAGNOSIS FROM LABORATORY INNER JOIN PATIENT ON LABORATORY.ID = PATIENT.ID INNER JOIN EXAMINATION ON LABORATORY.ID = EXAMINATION.ID ORDER BY LABORATORY.HGB DESC LIMIT 1;
-- ----------sql2------------
-- 0, 
SELECT STRFTIME('%Y', T2.DATE) - STRFTIME('%Y', T1.BIRTHDAY), T1.DIAGNOSIS FROM PATIENT AS T1 INNER JOIN LABORATORY AS T2 ON T1.ID = T2.ID ORDER BY T2.HGB DESC LIMIT 1;
