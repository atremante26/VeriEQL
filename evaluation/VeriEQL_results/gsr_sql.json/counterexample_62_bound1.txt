);

CREATE TABLE SCHOOLS (
	CDSCODE VARCHAR(20),
	COUNTY VARCHAR(20),
	SCHOOL VARCHAR(20),
	CHARTER INTEGER
);

-- ----------sql1------------
-- 1
SELECT COUNT(*) FROM SCHOOLS S JOIN FRPM F ON S.CDSCODE = F.CDSCODE WHERE S.CHARTER = 0 AND S.COUNTY = 'LOS ANGELES' AND (CAST(F.FREE_MEAL_COUNT_K12 AS FLOAT) * 100.0 / CAST(F.ENROLLMENT_K12 AS FLOAT)) < 0.18;
-- ----------sql2------------
-- 0
SELECT COUNT(T2.SCHOOL) FROM FRPM AS T1 INNER JOIN SCHOOLS AS T2 ON T1.CDSCODE = T2.CDSCODE WHERE T2.COUNTY = 'LOS ANGELES' AND T2.CHARTER = 0 AND CAST(T1.FREE_MEAL_COUNT_K12 AS REAL) * 100 / T1.ENROLLMENT_K12 < 0.18;
