SELECT CAST(STRFTIME('%Y', E.EXAMINATION_DATE) AS INTEGER) - CAST(STRFTIME('%Y', P.BIRTHDAY) AS INTEGER) AS AGE, E.DIAGNOSIS FROM EXAMINATION E JOIN PATIENT P ON E.ID = P.ID JOIN LABORATORY L ON E.ID = L.ID AND E.EXAMINATION_DATE = L.DATE ORDER BY L.HGB DESC LIMIT 1;
